# 缓存优化技巧

## 概念介绍

缓存优化是前端性能优化的重要技术，通过将计算结果存储在内存中，避免重复计算相同的输入，从而显著提升函数执行效率。这种技术特别适用于递归算法、复杂计算和API调用等场景。

## 基本原理

缓存优化的核心思想是**空间换时间**：
- 第一次计算时，将结果存储到缓存中
- 后续相同输入直接从缓存返回结果
- 避免重复计算，提高执行效率

## 四种主要实现方式

### 1. 通用memoize函数

最通用和灵活的实现方式，可以为任何函数添加缓存功能：

```javascript
/**
 * @description 通用记忆化函数，为任何函数添加缓存功能
 * @param {Function} fn - 需要缓存的原函数
 * @returns {Function} 带缓存功能的新函数
 */
function memoize(fn) {
  const cache = new Map();

  return function (...args) {
    // 将参数序列化作为缓存键
    const key = JSON.stringify(args);

    // 缓存命中，直接返回
    if (cache.has(key)) {
      console.log(`缓存命中: ${key}`);
      return cache.get(key);
    }

    // 缓存未命中，计算并存储
    console.log(`计算中: ${key}`);
    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}

// 应用示例：斐波那契数列
const fibonacci = memoize(function(n) {
  if (n <= 1) return n;
  return fibonacci(n - 1) + fibonacci(n - 2);
});

// 测试执行
console.log(fibonacci(10)); // 第一次计算
console.log(fibonacci(10)); // 从缓存获取
console.log(fibonacci(8));  // 部分从缓存获取

// 应用示例：复杂计算函数
const expensiveCalculation = memoize(function(x, y, z) {
  console.log('执行复杂计算...');
  return Math.pow(x, y) + Math.sqrt(z);
});

console.log(expensiveCalculation(2, 10, 16)); // 1028
console.log(expensiveCalculation(2, 10, 16)); // 从缓存返回 1028
```

### 2. 闭包封装私有缓存

使用闭包创建私有缓存空间，外部无法直接访问和修改缓存：

```javascript
/**
 * @description 使用闭包封装的私有缓存斐波那契函数
 */
const fibonacciClosure = (function() {
  // 私有缓存，外部无法直接访问
  const cache = new Map();

  return function(n) {
    // 检查缓存
    if (cache.has(n)) {
      console.log(`闭包缓存命中: fibonacci(${n})`);
      return cache.get(n);
    }

    // 基本情况
    if (n <= 1) {
      const result = n;
      cache.set(n, result);
      return result;
    }

    // 递归计算并缓存
    console.log(`闭包计算: fibonacci(${n})`);
    const result = fibonacciClosure(n - 1) + fibonacciClosure(n - 2);
    cache.set(n, result);
    return result;
  };
})();

// 工厂函数版本 - 可以创建多个独立的缓存实例
function createCachedFibonacci() {
  const cache = new Map();

  return function fibonacci(n) {
    if (cache.has(n)) return cache.get(n);

    if (n <= 1) {
      cache.set(n, n);
      return n;
    }

    const result = fibonacci(n - 1) + fibonacci(n - 2);
    cache.set(n, result);
    return result;
  };
}

// 创建两个独立的缓存实例
const fib1 = createCachedFibonacci();
const fib2 = createCachedFibonacci();

console.log(fib1(10)); // 55
console.log(fib2(10)); // 55（独立计算和缓存）
```

### 3. 函数属性缓存

直接在函数对象上添加缓存属性，简单直接：

```javascript
/**
 * @description 使用函数属性实现缓存的斐波那契函数
 * @param {number} n - 斐波那契数列位置
 * @returns {number} 斐波那契数值
 */
function fibonacciProperty(n) {
  // 初始化缓存（如果不存在）
  if (!fibonacciProperty.cache) {
    fibonacciProperty.cache = new Map();
  }

  // 检查缓存
  if (fibonacciProperty.cache.has(n)) {
    console.log(`属性缓存命中: fibonacci(${n})`);
    return fibonacciProperty.cache.get(n);
  }

  // 基本情况
  if (n <= 1) {
    fibonacciProperty.cache.set(n, n);
    return n;
  }

  // 递归计算
  console.log(`属性缓存计算: fibonacci(${n})`);
  const result = fibonacciProperty(n - 1) + fibonacciProperty(n - 2);
  fibonacciProperty.cache.set(n, result);
  return result;
}

// 可以访问和清理缓存
console.log(fibonacciProperty(8)); // 21
console.log('缓存大小:', fibonacciProperty.cache.size);

// 清理缓存
fibonacciProperty.clearCache = function() {
  this.cache.clear();
  console.log('缓存已清理');
};

fibonacciProperty.clearCache();
```

## 高级缓存技巧

### 1. 带过期时间的缓存

```javascript
/**
 * @description 带过期时间的缓存实现
 * @param {Function} fn - 原函数
 * @param {number} ttl - 缓存存活时间（毫秒）
 */
function memoizeWithTTL(fn, ttl = 60000) {
  const cache = new Map();

  return function(...args) {
    const key = JSON.stringify(args);
    const cached = cache.get(key);

    // 检查缓存是否过期
    if (cached && Date.now() - cached.timestamp < ttl) {
      console.log(`TTL缓存命中: ${key}`);
      return cached.value;
    }

    // 计算新结果
    console.log(`TTL缓存计算: ${key}`);
    const result = fn.apply(this, args);
    cache.set(key, {
      value: result,
      timestamp: Date.now()
    });

    return result;
  };
}

// 示例：API调用缓存（5秒过期）
const fetchUserData = memoizeWithTTL(async function(userId) {
  console.log(`请求用户数据: ${userId}`);
  // 模拟API调用
  return new Promise(resolve => {
    setTimeout(() => {
      resolve({ id: userId, name: `用户${userId}` });
    }, 1000);
  });
}, 5000);

// 测试
fetchUserData(123).then(console.log);
setTimeout(() => fetchUserData(123).then(console.log), 3000); // 缓存命中
setTimeout(() => fetchUserData(123).then(console.log), 7000); // 重新请求
```

### 2. LRU（最近最少使用）缓存

```javascript
/**
 * @description LRU缓存实现
 */
class LRUCache {
  constructor(capacity = 100) {
    this.capacity = capacity;
    this.cache = new Map();
  }

  get(key) {
    if (this.cache.has(key)) {
      // 更新访问顺序
      const value = this.cache.get(key);
      this.cache.delete(key);
      this.cache.set(key, value);
      return value;
    }
    return null;
  }

  set(key, value) {
    if (this.cache.has(key)) {
      // 更新现有键
      this.cache.delete(key);
    } else if (this.cache.size >= this.capacity) {
      // 删除最久未使用的项
      const firstKey = this.cache.keys().next().value;
      this.cache.delete(firstKey);
    }

    this.cache.set(key, value);
  }

  clear() {
    this.cache.clear();
  }

  size() {
    return this.cache.size;
  }
}

// 带LRU的memoize函数
function memoizeWithLRU(fn, capacity = 100) {
  const cache = new LRUCache(capacity);

  return function(...args) {
    const key = JSON.stringify(args);
    const cached = cache.get(key);

    if (cached !== null) {
      console.log(`LRU缓存命中: ${key}`);
      return cached;
    }

    console.log(`LRU缓存计算: ${key}`);
    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}
```

### 3. 多级缓存策略

```javascript
/**
 * @description 多级缓存：内存 + LocalStorage
 */
function createMultiLevelCache(fn, options = {}) {
  const {
    memoryCapacity = 100,
    useLocalStorage = true,
    keyPrefix = 'cache_'
  } = options;

  const memoryCache = new LRUCache(memoryCapacity);

  return function(...args) {
    const key = JSON.stringify(args);
    const storageKey = keyPrefix + key;

    // 1. 检查内存缓存
    let result = memoryCache.get(key);
    if (result !== null) {
      console.log(`内存缓存命中: ${key}`);
      return result;
    }

    // 2. 检查LocalStorage缓存
    if (useLocalStorage) {
      try {
        const stored = localStorage.getItem(storageKey);
        if (stored) {
          result = JSON.parse(stored);
          memoryCache.set(key, result); // 回写到内存缓存
          console.log(`LocalStorage缓存命中: ${key}`);
          return result;
        }
      } catch (e) {
        console.warn('LocalStorage读取失败:', e);
      }
    }

    // 3. 计算新结果
    console.log(`多级缓存计算: ${key}`);
    result = fn.apply(this, args);

    // 4. 存储到两级缓存
    memoryCache.set(key, result);
    if (useLocalStorage) {
      try {
        localStorage.setItem(storageKey, JSON.stringify(result));
      } catch (e) {
        console.warn('LocalStorage写入失败:', e);
      }
    }

    return result;
  };
}
```

## 缓存策略选择指南

### 使用场景对比

| 方式 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| **memoize函数** | 通用性强，可复用 | 额外函数调用开销 | 需要为多个函数添加缓存 |
| **闭包缓存** | 私有性好，性能佳 | 不够灵活，一次性 | 单个函数需要私有缓存 |
| **函数属性** | 简单直接，可访问缓存 | 缓存暴露，可能冲突 | 需要手动管理缓存 |
| **全局变量** | 实现简单，跨函数共享 | 污染全局，命名冲突 | 简单应用，临时使用 |

### 选择建议

1. **通用需求**：优先选择`memoize`函数
2. **性能关键**：使用闭包缓存
3. **需要缓存控制**：使用函数属性
4. **简单场景**：可以考虑全局变量（加命名空间）
5. **复杂应用**：考虑多级缓存或专业缓存库

## 注意事项和最佳实践

### 1. 内存管理
```javascript
// 定期清理缓存，避免内存泄漏
function createManagedCache(fn, maxAge = 300000) { // 5分钟
  const cache = new Map();

  // 定期清理过期缓存
  setInterval(() => {
    const now = Date.now();
    for (const [key, value] of cache) {
      if (now - value.timestamp > maxAge) {
        cache.delete(key);
      }
    }
  }, maxAge / 2);

  return function(...args) {
    const key = JSON.stringify(args);
    const cached = cache.get(key);

    if (cached && Date.now() - cached.timestamp < maxAge) {
      return cached.value;
    }

    const result = fn.apply(this, args);
    cache.set(key, {
      value: result,
      timestamp: Date.now()
    });

    return result;
  };
}
```

```javascript
// 自定义键生成函数，处理复杂参数
```

## 面试常见问题

### 问题1：什么是记忆化（Memoization）？它的原理是什么？

**答**：
记忆化是一种优化技术，通过缓存函数的计算结果来避免重复计算。原理是：
1. 第一次调用函数时，计算结果并存储在缓存中
2. 后续相同参数的调用直接从缓存返回结果
3. 用空间换时间，特别适合递归和计算密集型函数

核心优势：将指数时间复杂度降为线性时间复杂度。

### 问题2：比较不同缓存实现方式的优缺点

**答**：
- **memoize函数**：通用性最强，但有函数调用开销
- **闭包缓存**：性能最佳，私有性好，但不够灵活
- **函数属性**：实现简单，可以访问和控制缓存，但缓存暴露
- **全局变量**：最简单，但污染全局命名空间

选择依据：通用性需求 → memoize；性能要求 → 闭包；控制需求 → 函数属性。

### 问题3：缓存什么时候会失效？如何处理？

**答**：
缓存失效场景：
1. **数据过期**：设置TTL时间自动失效
2. **内存限制**：使用LRU策略清理旧数据
3. **参数变化**：不同参数产生不同缓存键
4. **手动清理**：提供清理接口

处理方案：TTL + LRU + 手动清理 + 多级缓存策略。

### 问题4：大型应用中如何设计缓存架构？

**答**：
1. **分层缓存**：内存缓存 + 本地存储 + 服务端缓存
2. **缓存策略**：按业务重要性设置不同TTL
3. **缓存清理**：定期清理 + LRU策略 + 手动清理接口
4. **监控报警**：缓存命中率监控 + 内存使用监控
5. **容错机制**：缓存失败时的降级方案

关键是根据业务特点选择合适的缓存策略和清理机制。

## 常见错误分析与复杂度对比

### 递归函数缓存实现的重大陷阱

在实现递归函数缓存时，有一个极其容易犯的错误，会导致缓存完全失效。让我们通过具体案例来分析：

#### 错误写法分析

```javascript
/**
 * @description 错误的递归缓存实现 - 缓存失效
 * ❌ 这种写法看似正确，实际上递归调用不会使用缓存！
 */
const fibonacciWrong = memoize(function fibonacci(n) {
  console.log(`内部调用: fibonacci(${n})`);
  if (n <= 1) return n;
  // ❌ 问题：这里的fibonacci是函数内部名称，不是外部memoize包装后的版本
  return fibonacci(n - 1) + fibonacci(n - 2);
});

/**
 * @description 正确的递归缓存实现
 * ✅ 递归调用会使用缓存
 */
const fibonacciCorrect = memoize(function(n) {
  console.log(`内部调用: 匿名函数(${n})`);
  if (n <= 1) return n;
  // ✅ 正确：这里的fibonacciCorrect是外部变量，指向memoize包装后的函数
  return fibonacciCorrect(n - 1) + fibonacciCorrect(n - 2);
});
```

#### 问题根源解析

**核心问题**：JavaScript中函数的内部名称和外部变量是不同的引用

```javascript
// 详细分析错误写法的执行过程
const fibonacciWrong = memoize(function fibonacci(n) {
  if (n <= 1) return n;
  // 这里的fibonacci指向的是：
  // function fibonacci(n) { ... } (原始函数，未包装)
  // 而不是：
  // memoize(function fibonacci(n) { ... }) (包装后的函数)
  return fibonacci(n - 1) + fibonacci(n - 2);
});

// 调用链分析
fibonacciWrong(5); // 外层调用使用缓存
  ↓
function fibonacci(5) { // 内部递归不使用缓存！
  return fibonacci(4) + fibonacci(3); // 这些调用都是原始函数
    ↓
  function fibonacci(4) { // 继续不使用缓存
    return fibonacci(3) + fibonacci(2);
    // ... 大量重复计算
  }
}
```

```
内部调用: fibonacci(6)  // 重复计算
内部调用: fibonacci(5)  // 重复计算
// ... 总共调用 177 次！
```

**正确写法执行 `fibonacciCorrect(10)`**：
```
缓存未命中: [10]
内部调用: 匿名函数(10)
缓存未命中: [9]
内部调用: 匿名函数(9)
缓存未命中: [8]
内部调用: 匿名函数(8)
// ... 每个数字只计算一次，总共调用 11 次
```

### 复杂度分析对比

#### 1. 无缓存原始递归

```javascript
function fibonacciNaive(n) {
  if (n <= 1) return n;
  return fibonacciNaive(n - 1) + fibonacciNaive(n - 2);
}
```

**时间复杂度**：O(2^n)
- 每个函数调用会产生两个子调用
- 调用次数呈指数级增长
- fibonacci(n) ≈ 2^n 次函数调用

**空间复杂度**：O(n)
- 递归深度最大为 n
- 调用栈空间消耗为 O(n)

**具体调用次数**：
```javascript
/**
 * @description 统计原始递归的调用次数
 */

  fibonacciNaiveWithCount.count = (fibonacciNaiveWithCount.count || 0) + 1;
  if (n <= 1) return n;
  return fibonacciNaiveWithCount(n - 1) + fibonacciNaiveWithCount(n - 2);
}

// 测试调用次数
fibonacciNaiveWithCount.count = 0;
console.log(`fibonacci(10) = ${fibonacciNaiveWithCount(10)}`);
console.log(`调用次数: ${fibonacciNaiveWithCount.count}`); // 177 次

fibonacciNaiveWithCount.count = 0;
console.log(`fibonacci(20) = ${fibonacciNaiveWithCount(20)}`);
console.log(`调用次数: ${fibonacciNaiveWithCount.count}`); // 21891 次

fibonacciNaiveWithCount.count = 0;
console.log(`fibonacci(30) = ${fibonacciNaiveWithCount(30)}`);
console.log(`调用次数: ${fibonacciNaiveWithCount.count}`); // 2692537 次
```

#### 2. 错误缓存实现（递归不使用缓存）

```javascript
const fibonacciWrongCached = memoize(function fibonacci(n) {
  if (n <= 1) return n;
  return fibonacci(n - 1) + fibonacci(n - 2); // 内部不使用缓存
});
```

**时间复杂度**：O(2^n) - 与无缓存相同！
- 只有最外层调用会检查缓存
- 内部递归调用仍然是指数级
- 缓存几乎没有效果

**空间复杂度**：O(n + k)
- 递归栈：O(n)
- 缓存空间：O(k)，k为不同参数的数量

#### 3. 正确缓存实现（递归使用缓存）

```javascript
const fibonacciCorrectCached = memoize(function(n) {
  if (n <= 1) return n;
  return fibonacciCorrectCached(n - 1) + fibonacciCorrectCached(n - 2);
});
```

**时间复杂度**：O(n)
- 每个子问题只计算一次
- fibonacci(0) 到 fibonacci(n) 各计算一次
- 总共 n+1 次计算

**空间复杂度**：O(n)
- 递归栈：O(n)
- 缓存空间：O(n)，存储 n+1 个结果

**调用次数验证**：
```javascript
/**
 * @description 验证正确缓存的调用次数
 */
function createCountedMemoizedFib() {
  let count = 0;
  const fib = memoize(function(n) {
    count++;
    console.log(`计算 fibonacci(${n}), 总调用次数: ${count}`);
    if (n <= 1) return n;
    return fib(n - 1) + fib(n - 2);
  });
  fib.getCount = () => count;
  fib.resetCount = () => count = 0;
  return fib;
}

const countedFib = createCountedMemoizedFib();
console.log(`fibonacci(10) = ${countedFib(10)}`);
console.log(`总调用次数: ${countedFib.getCount()}`); // 11 次

countedFib.resetCount();
console.log(`fibonacci(20) = ${countedFib(20)}`);
console.log(`总调用次数: ${countedFib.getCount()}`); // 21 次

countedFib.resetCount();
console.log(`fibonacci(30) = ${countedFib(30)}`);
console.log(`总调用次数: ${countedFib.getCount()}`); // 31 次
```

### 正确实现方案

#### 方案1：使用外部变量名（推荐）

```javascript
const fibonacci = memoize(function(n) {
  if (n <= 1) return n;
  return fibonacci(n - 1) + fibonacci(n - 2); // ✅ 使用外部变量
});
```

#### 方案2：显式传递包装后的函数

```javascript
function createMemoizedFibonacci() {
  const fibonacci = memoize(function(n) {
    if (n <= 1) return n;
    return fibonacci(n - 1) + fibonacci(n - 2); // ✅ 引用外部变量
  });
  return fibonacci;
}
```

#### 方案3：传递自身引用参数

```javascript
const fibonacci = memoize(function fib(n, self = fibonacci) {
  if (n <= 1) return n;
  return self(n - 1) + self(n - 2); // ✅ 使用传入的self参数
});
```

### 关键总结

1. **命名函数陷阱**：内部函数名与外部变量是不同引用
2. **性能差异巨大**：错误实现可能慢数万倍
3. **复杂度本质**：O(2^n) vs O(n) 的根本差异
4. **最佳实践**：使用匿名函数 + 外部变量引用
5. **测试重要性**：必须验证递归调用是否真正使用缓存

这个错误在面试中经常出现，是考察候选人对JavaScript闭包、变量作用域和性能优化深度理解的经典问题。
